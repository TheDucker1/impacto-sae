add_library(saekano_core STATIC formats.cpp)
target_include_directories(saekano_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(saekano_core PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(saekano_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(saekano_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(saekano_inspect inspect.cpp)
target_link_libraries(saekano_inspect PRIVATE saekano_core)
add_executable(saekano_core_tests ../../tests/saekano/formats_test.cpp)
target_link_libraries(saekano_core_tests PRIVATE saekano_core)
add_test(NAME saekano_formats COMMAND saekano_core_tests)
